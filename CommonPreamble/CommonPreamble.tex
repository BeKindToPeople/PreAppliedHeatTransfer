% --- Custom TikZ Definitions for Equation Box ---
\makeatletter
% 1. Setup Keys
\pgfkeys{
 /tikz/eqnbox/.cd,
 tri width/.store in=\triW, tri width=12pt,
 tri height/.store in=\triH, tri height=10pt,
 box color/.store in=\boxCol, box color=blue!80!black,
 fill color/.store in=\fillCol, fill color=blue!5,
}

% 2. Define the "Two Triangles" Top Edge Decoration
\pgfdeclaredecoration{two triangles on top}{initial}{
  \state{initial}[width=\pgfdecoratedpathlength]{
    \pgfmathsetlengthmacro{\halfTriW}{\triW/2}
    \pgfmathsetlengthmacro{\OneThirdLen}{\pgfdecoratedpathlength/3}
    \pgfmathsetlengthmacro{\TwoThirdLen}{2*\pgfdecoratedpathlength/3}
    \pgfpathmoveto{\pgfpoint{0pt}{0pt}}
    \pgfpathlineto{\pgfpoint{\OneThirdLen - \halfTriW}{0pt}}
    \pgfpathlineto{\pgfpoint{\OneThirdLen}{\triH}}
    \pgfpathlineto{\pgfpoint{\OneThirdLen + \halfTriW}{0pt}}
    \pgfpathlineto{\pgfpoint{\TwoThirdLen - \halfTriW}{0pt}}
    \pgfpathlineto{\pgfpoint{\TwoThirdLen}{\triH}}
    \pgfpathlineto{\pgfpoint{\TwoThirdLen + \halfTriW}{0pt}}
    \pgfpathlineto{\pgfpoint{\pgfdecoratedpathlength}{0pt}}
  }
  \state{final}{}
}

% 3. Define the Style
\tikzset{
  eqn box/.style={
    /tikz/eqnbox/.cd, #1, /tikz/.cd,
    shape=rectangle, inner sep=12pt, fill=\fillCol, draw=none,
    drop shadow={opacity=0.2, shadow xshift=2pt, shadow yshift=-2pt},
    append after command={
      \pgfextra{
        \draw[thick, \boxCol, line join=round, rounded corners=3pt]
          (\tikzlastnode.north east) -- (\tikzlastnode.south east) --
          (\tikzlastnode.south west) -- (\tikzlastnode.north west);
        \draw[thick, \boxCol, line join=miter, sharp corners,
              decorate, decoration={two triangles on top}]
          (\tikzlastnode.north west) -- (\tikzlastnode.north east);
      }
    }
  }
}
